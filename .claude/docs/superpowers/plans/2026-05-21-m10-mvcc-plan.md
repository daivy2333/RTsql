```rust
    /// Traverse version chain to find first visible version (M10)
    ///
    /// Returns: visible tuple bytes, or None if all versions invisible
    pub async fn find_visible_version(
        &self,
        row_id: RowId,
        snapshot: &Snapshot,
    ) -> Result<Option<Vec<u8>>> {
        let mut current_row_id = Some(row_id);

        while let Some(current) = current_row_id {
            // Read current version from data page
            let (version_header, tuple_bytes) =
                crate::storage::read_tuple_from_data_page(self, current).await?;

            // Check visibility
            let create_tx = version_header.create_tx_id();
            let commit_tx = version_header.commit_tx_id();

            let visible = snapshot.is_visible(create_tx, commit_tx)
                || snapshot.is_visible_self(create_tx, commit_tx);

            if visible {
                return Ok(Some(tuple_bytes));
            }

            // Not visible, follow next_version pointer
            current_row_id = version_header.next_version();
        }

        // All versions invisible
        Ok(None)
    }
```

```rust
/// Update version header in an existing data page slot (M10)
///
/// Used by write_commit_tx_id to update commit marker without changing tuple data
pub async fn update_version_header_in_data_page(
    buffer_pool: &BufferPool,
    row_id: RowId,
    new_header: VersionHeader,
    tuple_bytes: &[u8],
) -> Result<()> {
    // Read the page
    let page_id = row_id.page_id();
    let slot_id = row_id.slot_id();

    let mut page_guard = buffer_pool.read_page(page_id).await?;
    let page = page_guard.data_mut();

    // Reconstruct full row data: header + tuple
    let header_bytes = new_header.to_bytes();
    let total_size = VersionHeader::SIZE + tuple_bytes.len();

    // Overwrite the slot data
    // Note: This assumes slot size unchanged (which is true for commit marker)
    let slotted = SlottedPage::parse(page)?;
    let slot_offset = slotted.slot_offset(slot_id)?;
    let slot_end = slot_offset + total_size;

    // Update header bytes
    page[slot_offset..slot_offset + VersionHeader::SIZE].copy_from_slice(&header_bytes);

    // Mark page as dirty
    page_guard.mark_dirty();

    Ok(())
}
```

需要在 `data_page.rs` 顶部添加 `use crate::storage::page_format::{VersionHeader, SlottedPage};`

```rust
    /// Commit a transaction
    ///
    /// - Marks all versions as committed (M10)
    /// - Removes from active list
    pub async fn commit(&self, tx: Transaction, buffer_pool: &BufferPool) -> Result<()> {
        let tx_id = tx.id();

        // M10: Mark all versions as committed
        self.commit_mark_versions(tx_id, buffer_pool).await?;

        // Remove from active list
        let mut active = self.active_tx_ids.write().await;
        if !active.remove(&tx_id) {
            return Err(TransactionError::AlreadyCommitted(tx_id));
        }

        // Clear tx_versions for this transaction
        self.tx_versions.write().await.remove(&tx_id);

        Ok(())
    }
```

```rust
    /// Cleanup uncommitted versions during abort (M10)
    pub async fn abort_cleanup_versions(
        &self,
        tx_id: u64,
        buffer_pool: &BufferPool,
        table_meta: &TableMeta,
    ) -> Result<()> {
        let versions = self.tx_versions.read().await;
        let tx_versions = versions.get(&tx_id).cloned().unwrap_or_default();

        for row_id in tx_versions {
            // Read version header to get next_version
            let header = buffer_pool.read_version_header(row_id).await?;

            // Find the key for this row_id
            let key = table_meta.index_manager.find_key_by_row_id(row_id).await;

            if let Some(key) = key {
                // This version is pointed by index
                // Restore index to next_version if exists
                if let Some(prev_row_id) = header.next_version() {
                    table_meta.index_manager.update(&key, prev_row_id).await?;
                } else {
                    // No previous version, remove index entry
                    table_meta.index_manager.delete(&key).await?;
                }
            }
        }

        Ok(())
    }
```

```rust
    pub async fn gc_table(
        &self,
        buffer_pool: &BufferPool,
    ) -> Result<usize> {
        let mut cleaned_count = 0;

        let all_entries = self.index_manager.scan_all_entries().await?;

        for (key, row_id) in all_entries {
            let mut current = Some(row_id);
            let mut prev_versions = Vec::new();

            // Traverse version chain, collect old versions
            while let Some(current_id) = current {
                let header = buffer_pool.read_version_header(current_id).await?;

                // Collect committed old versions (not latest)
                if header.commit_tx_id().is_some() && current_id != row_id {
                    prev_versions.push(current_id);
                }

                current = header.next_version();
            }

            // Delete old versions
            for old_id in prev_versions {
                crate::storage::delete_tuple_from_data_page(buffer_pool, old_id).await?;
                cleaned_count += 1;
            }
        }

        Ok(cleaned_count)
    }
```
